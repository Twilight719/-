# 罗德岛群聊室功能完整实现方案（KimiCode版）

---

## 一、项目概述

在现有单聊App基础上，增加**群聊模式**。多个AI角色（阿米娅、凯尔希、Mon3tr、可露希尔、德克萨斯、能天使等）在同一个群聊频道中互动。用户作为"博士"发起话题，AI角色根据话题相关性、性格、关系网自动决定谁说话、谁沉默、谁接话。

```typescript
// services/groupOrchestrator.ts

import { GroupMember, GroupMessage } from '../types/groupChat';

interface SpeakerScore {
  member: GroupMember;
  score: number;
  reasons: string[];
}

export function decideResponders(
  userMessage: string,
  members: GroupMember[],
  history: GroupMessage[],
  maxResponders: number = 2
): { speakerId: string; reasoning: string }[] {

  const scores: SpeakerScore[] = members.map(m => {
    let score = 0;
    const reasons: string[] = [];

    // 1. 被直接点名（权重50%）
    if (userMessage.includes(m.name)) {
      score += 50;
      reasons.push('被博士点名');
    }

    // 2. 关键词匹配（权重40%）
    const matched = m.triggerWords.filter(w => 
      userMessage.toLowerCase().includes(w.toLowerCase())
    );
    if (matched.length > 0) {
      score += matched.length * 15;
      reasons.push(`关键词匹配(${matched.join(',')})`);
    }

    // 3. 性格活跃度（权重10%）
    const activityScore = (1 - m.silenceThreshold) * 10;
    score += activityScore;
    if (activityScore > 5) reasons.push('性格活跃');

    // 4. 历史发言频率惩罚（避免某人话太多）
    const recentCount = history.slice(-8).filter(h => h.senderId === m.id).length;
    const penalty = recentCount * 8;
    score -= penalty;
    if (penalty > 0) reasons.push(`话多惩罚(-${penalty})`);

    // 5. 话题热度加成
    const topicHeat = calculateTopicHeat(userMessage, m);
    score += topicHeat;
    if (topicHeat > 0) reasons.push(`话题热度(+${topicHeat})`);

    return { member: m, score: Math.max(0, score), reasons };
  });

  // 排序：分数高的优先
  scores.sort((a, b) => b.score - a.score);

  const results: { speakerId: string; reasoning: string }[] = [];

  // 选主回应者（分数最高的，必须>20分）
  if (scores[0].score > 20) {
    results.push({
      speakerId: scores[0].member.id,
      reasoning: `主回应：${scores[0].reasons.join(' | ')}(${scores[0].score}分)`
    });
  }

  // 选副回应者（分数第二，且与主回应者分差<30，避免不相关的人硬凑）
  if (scores.length > 1 && scores[1].score > 15 && (scores[0].score - scores[1].score) < 30) {
    results.push({
      speakerId: scores[1].member.id,
      reasoning: `副回应：${scores[1].reasons.join(' | ')}(${scores[1].score}分)`
    });
  }

  // 如果没人达标，随机选一个活跃角色保底
  if (results.length === 0) {
    const active = members.filter(m => m.silenceThreshold < 0.6);
    const random = active[Math.floor(Math.random() * active.length)];
    results.push({
      speakerId: random.id,
      reasoning: '保底随机：话题不匹配，选活跃角色接话'
    });
  }

  return results;
}

// AI之间互聊的调度
export function decideNextSpeaker(
  lastMessage: GroupMessage,
  history: GroupMessage[],
  members: GroupMember[]
): { speakerId: string; reasoning: string } | null {

  const lastContent = lastMessage.content;
  const lastSender = members.find(m => m.id === lastMessage.senderId);

  // 1. 被点名必回
  const mentioned = members.find(m => 
    lastContent.includes(m.name) && m.id !== lastMessage.senderId
  );
  if (mentioned && Math.random() > 0.2) { // 80%概率回
    return {
      speakerId: mentioned.id,
      reasoning: `被${lastSender?.name}点名，回应`
    };
  }

  // 2. 关系驱动的互动（制造群聊感）
  const relationshipTriggers: Record<string, Array<{target: string; prob: number; reason: string}>> = {
    'exusiai': [{ target: 'texas', prob: 0.6, reason: '能天使说完，德克萨斯吐槽' }],
    'texas': [{ target: 'exusiai', prob: 0.4, reason: '德克萨斯说完，能天使接茬' }],
    'closure': [{ target: 'kaltsit', prob: 0.5, reason: '可露希尔说完，凯尔希镇压' }],
    'kaltsit': [{ target: 'closure', prob: 0.3, reason: '凯尔希说完，可露希尔秒怂' }],
    'amiya': [{ target: 'kaltsit', prob: 0.3, reason: '阿米娅说完，凯尔希关心' }],
    'w': [{ target: 'kaltsit', prob: 0.4, reason: 'W说完，凯尔希警告' }],
  };

  const triggers = relationshipTriggers[lastMessage.senderId];
  if (triggers) {
    for (const t of triggers) {
      if (Math.random() < t.prob) {
        return {
          speakerId: t.target,
          reasoning: t.reason
        };
      }
    }
  }

  // 3. 话题延续（30%概率随机接话）
  if (Math.random() > 0.7) {
    const others = members.filter(m => m.id !== lastMessage.senderId);
    const random = others[Math.floor(Math.random() * others.length)];
    return {
      speakerId: random.id,
      reasoning: '话题延续，随机接话'
    };
  }

  // 4. 默认沉默（不触发互聊）
  return null;
}

// 话题热度计算
function calculateTopicHeat(message: string, member: GroupMember): number {
  const heatMap: Record<string, { heat: number; members: string[] }> = {
    '任务': { heat: 15, members: ['amiya', 'kaltsit', 'texas'] },
    '炸弹': { heat: 20, members: ['w', 'closure'] },
    'pocky': { heat: 15, members: ['texas', 'exusiai'] },
    '凯尔希': { heat: 20, members: ['mon3tr', 'closure', 'kaltsit'] },
    '休息': { heat: 10, members: ['kaltsit', 'amiya'] },
    '买东西': { heat: 25, members: ['closure'] },
    '苹果派': { heat: 15, members: ['exusiai'] },
    '医疗': { heat: 15, members: ['kaltsit', 'amiya'] },
    '工程': { heat: 15, members: ['closure'] },
    '叙拉古': { heat: 20, members: ['texas', 'lappland'] },
    '拉特兰': { heat: 15, members: ['exusiai'] },
    '卡西米尔': { heat: 15, members: ['nearl'] },
  };

  let totalHeat = 0;
  for (const [keyword, data] of Object.entries(heatMap)) {
    if (message.includes(keyword) && data.members.includes(member.id)) {
      totalHeat += data.heat;
    }
  }
  return totalHeat;
}
```

```typescript
// services/groupContext.ts

import { GroupMember, GroupMessage } from '../types/groupChat';

export function buildGroupSystemPrompt(
  member: GroupMember,
  allMembers: GroupMember[],
  history: GroupMessage[]
): string {

  const otherMembers = allMembers.filter(m => m.id !== member.id);

  const memberList = otherMembers.map(m => {
    const relation = member.relationships[m.id] || '普通同事';
    return `- ${m.name}（${m.personality}）：你对TA的态度是"${relation}"`;
  }).join('
');

  const recentHistory = history.slice(-8).map(h => {
    const prefix = h.isUser ? '博士' : h.senderName;
    return `${prefix}：${h.content}`;
  }).join('
');

  return `
【系统指令：群聊模式】
你正在参与一个名为"罗德岛内部通讯频道"的群聊。
你是${member.name}，${member.personality}。

【当前群聊成员】
${memberList}

【群聊规则】
1. 这是群聊，你的回复应该像群聊消息一样自然简短（2-3句话）
2. 你可以@其他成员（用@名字），也可以直接说话
3. 你可以接话、吐槽、打断、或保持沉默
4. 群聊氛围轻松，允许开玩笑、互损，但保持角色性格
5. 如果博士（用户）发言，优先回应博士的话题
6. 如果其他AI刚说完，你可以根据内容选择接话或沉默
7. 不要重复其他角色刚说过的内容
8. 在群聊中比单聊更随意，可以用"~"和简单emoji
9. 如果${member.relationships['kaltsit'] ? '凯尔希' : '某个权威角色'}在群里，你会表现出对应态度
10. 回复控制在80字以内，保持聊天节奏轻快

【最近群聊记录】
${recentHistory || '（群聊刚开始）'}

现在请作为${member.name}回复。
`;
}
```

```typescript
// services/groupChat.ts

import { GroupMember, GroupMessage, ChatRound } from '../types/groupChat';
import { decideResponders, decideNextSpeaker } from './groupOrchestrator';
import { buildGroupSystemPrompt, buildAIChatContext } from './groupContext';
import { streamChat } from './deepseek';

const MAX_AI_CHAT_DEPTH = 2;      // AI互聊最多2轮
const MAX_HISTORY_LENGTH = 15;    // 保留最近15轮
const TYPING_DELAY_MIN = 500;     // 最小打字延迟ms
const TYPING_DELAY_MAX = 1500;    // 最大打字延迟ms

export class GroupChatController {
  private members: GroupMember[];
  private messages: GroupMessage[] = [];
  private typingMembers: Set<string> = new Set();
  private onMessageCallback?: (msg: GroupMessage) => void;
  private onTypingCallback?: (memberId: string, isTyping: boolean) => void;

  constructor(members: GroupMember[]) {
    this.members = members;
  }

  setCallbacks(
    onMessage: (msg: GroupMessage) => void,
    onTyping: (memberId: string, isTyping: boolean) => void
  ) {
    this.onMessageCallback = onMessage;
    this.onTypingCallback = onTyping;
  }

  // 用户发送消息
  async handleUserMessage(userText: string) {
    // 1. 记录用户消息
    const userMsg: GroupMessage = {
      id: `msg_${Date.now()}`,
      senderId: 'user',
      senderName: '博士',
      senderColor: '#FFFFFF',
      content: userText,
      timestamp: Date.now(),
      isUser: true,
    };
    this.addMessage(userMsg);

    // 2. 调度器决定谁回应
    const responders = decideResponders(userText, this.members, this.messages);

    // 3. 依次生成回复
    for (let i = 0; i < responders.length; i++) {
      const responder = responders[i];
      const member = this.members.find(m => m.id === responder.speakerId);
      if (!member) continue;

      // 显示正在输入
      this.setTyping(member.id, true);

      // 延迟（模拟打字速度）
      await this.delay(TYPING_DELAY_MIN + Math.random() * (TYPING_DELAY_MAX - TYPING_DELAY_MIN));

      // 构建上下文
      const systemPrompt = buildGroupSystemPrompt(member, this.members, this.messages);
      const history = this.getHistoryForMember(member.id);

      // 调用API
      try {
        const content = await this.generateResponse(
          systemPrompt,
          history,
          userText,
          member.model
        );

        // 发送消息
        const aiMsg: GroupMessage = {
          id: `msg_${Date.now()}_${i}`,
          senderId: member.id,
          senderName: member.name,
          senderColor: member.color,
          content,
          timestamp: Date.now(),
          isUser: false,
        };
        this.addMessage(aiMsg);
        this.setTyping(member.id, false);

        // 4. 触发AI互聊（最后一个回应者才触发）
        if (i === responders.length - 1) {
          await this.handleAIChatRound(aiMsg, 0);
        }
      } catch (error) {
        console.error(`[GroupChat] ${member.name}生成失败:`, error);
        this.setTyping(member.id, false);
      }
    }
  }

  // AI之间互聊
  private async handleAIChatRound(lastMsg: GroupMessage, depth: number) {
    if (depth >= MAX_AI_CHAT_DEPTH) return;

    const next = decideNextSpeaker(lastMsg, this.messages, this.members);
    if (!next) return;

    const member = this.members.find(m => m.id === next.speakerId);
    if (!member) return;

    // 显示正在输入
    this.setTyping(member.id, true);
    await this.delay(TYPING_DELAY_MIN + Math.random() * 1000);

    // 构建精简上下文
    const context = buildAIChatContext(
      member,
      this.members,
      this.messages,
      lastMsg.senderName,
      lastMsg.content
    );

    try {
      const content = await this.generateResponse(
        context,
        [],
        '',
        'deepseek-v4-flash' // 互聊用Flash，省钱
      );

      const aiMsg: GroupMessage = {
        id: `msg_${Date.now()}_ai_${depth}`,
        senderId: member.id,
        senderName: member.name,
        senderColor: member.color,
        content,
        timestamp: Date.now(),
        isUser: false,
      };
      this.addMessage(aiMsg);
      this.setTyping(member.id, false);

      // 递归触发下一轮（30%概率）
      if (Math.random() > 0.7) {
        await this.handleAIChatRound(aiMsg, depth + 1);
      }
    } catch (error) {
      this.setTyping(member.id, false);
    }
  }

  // 生成回复（流式）
  private async generateResponse(
    systemPrompt: string,
    history: Array<{role: string; content: string}>,
    userText: string,
    model: string
  ): Promise<string> {
    let content = '';
    const stream = streamChat(systemPrompt, history, userText, model);

    for await (const chunk of stream) {
      if (typeof chunk === 'string') {
        content += chunk;
      }
    }

    return content.trim();
  }

  // 工具方法
  private addMessage(msg: GroupMessage) {
    this.messages.push(msg);
    if (this.messages.length > MAX_HISTORY_LENGTH) {
      this.messages = this.messages.slice(-MAX_HISTORY_LENGTH);
    }
    this.onMessageCallback?.(msg);
  }

  private setTyping(memberId: string, isTyping: boolean) {
    if (isTyping) {
      this.typingMembers.add(memberId);
    } else {
      this.typingMembers.delete(memberId);
    }
    this.onTypingCallback?.(memberId, isTyping);
  }

  private getHistoryForMember(memberId: string): Array<{role: string; content: string}> {
    return this.messages.map(m => ({
      role: m.isUser ? 'user' : (m.senderId === memberId ? 'assistant' : 'user'),
      content: `${m.senderName}：${m.content}`
    }));
  }

  private delay(ms: number): Promise<void> {
    return new Promise(resolve => setTimeout(resolve, ms));
  }

  getMessages(): GroupMessage[] {
    return [...this.messages];
  }

  getTypingMembers(): string[] {
    return Array.from(this.typingMembers);
  }
}
```

## 十二、关键实现要点

1. **调度器是灵魂**：不是所有人都说话，而是"谁该说话"
2. **延迟制造真实感**：500-1500ms的打字延迟，多个指示器同时显示
3. **关系矩阵制造冲突**：德克萨斯↔能天使的损友互动是群聊的灵魂
4. **成本控制**：主回应Pro，接话Flash，互聊Flash，历史压缩
5. **群聊提示词改造**：每个角色增加"群聊环境感知"层
6. **话题热度系统**：让相关角色活跃，无关角色沉默
7. **保底机制**：如果话题不匹配，随机选活跃角色接话，避免冷场
